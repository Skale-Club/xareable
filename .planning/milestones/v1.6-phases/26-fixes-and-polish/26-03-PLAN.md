// drawBlocks() — the draw loop. NEVER sets ctx.font. THE BUG.
layouts.forEach((layout, i) => {
  const trackingPx = Math.round(layout.size_px * treatment.letterSpacingRatio);

    ctx.fillText(line, x, y);                  // <-- drawn with the WRONG font

    y += layout.line_height_px;                 // <-- advance IS correct (uses layout's own value)

});
